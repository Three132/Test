เชื่อมต่อ Google Sheet

เมื่อเชื่อมต่อแล้ว คะแนนที่บันทึกหรือแก้ไขในระบบจะถูกส่งไปยัง Google Sheet ของคุณโดยอัตโนมัติ

ขั้นตอนการติดตั้ง:

  1. เปิด Google Sheet ของคุณ หรือสร้างขึ้นใหม่
  2. กดเมนู ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเริ่มต้นออก แล้วคัดลอกโค้ดนี้ไปวาง:
    4. function doGet(e) { return handleRequest(e); }
function doPost(e) { return handleRequest(e); }

function handleRequest(e) {
  var category = e.parameter.category || "";
  var game = e.parameter.game || "1";
  var action = e.parameter.action || "";
  var name = e.parameter.name || "";
  var score = e.parameter.score || "0";
  var color = e.parameter.color || "";
  
  if (e.postData && e.postData.contents) {
    try {
      var data = JSON.parse(e.postData.contents);
      category = data.category || category;
      game = data.game || game;
      action = data.action || action;
      name = data.name || name;
      score = data.score || score;
      color = data.color || color;
    } catch(err) {}
  }
  
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var suffix = "Game " + game;
  var sheetName = category ? category + " - " + suffix : suffix;
  
  // Handle clear actions
  if (action === "clear") {
    var sheet = ss.getSheetByName(sheetName);
    if (sheet) {
      sheet.clearContents();
      sheet.appendRow(["เวลาบันทึก", "หมวดหมู่", "เกม", "ชื่อผู้เล่น", "คะแนน", "สี"]);
    }
    return ContentService.createTextOutput(JSON.stringify({status:"success"}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeaders({"Access-Control-Allow-Origin":"*"});
  }
  
  if (action === "clearAll") {
    for (var i = 1; i <= 4; i++) {
      var targetSuffix = "Game " + i;
      var targetSheetName = category ? category + " - " + targetSuffix : targetSuffix;
      var sheet = ss.getSheetByName(targetSheetName);
      if (sheet) {
        sheet.clearContents();
        sheet.appendRow(["เวลาบันทึก", "หมวดหมู่", "เกม", "ชื่อผู้เล่น", "คะแนน", "สี"]);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status:"success"}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeaders({"Access-Control-Allow-Origin":"*"});
  }
  
  // Default append row action
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["เวลาบันทึก", "หมวดหมู่", "เกม", "ชื่อผู้เล่น", "คะแนน", "สี"]);
  }
  
  if (name) {
    sheet.appendRow([new Date(), category, "Game " + game, name, Number(score), color]);
    return ContentService.createTextOutput(JSON.stringify({status:"success"}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeaders({"Access-Control-Allow-Origin":"*"});
  }
  
  return ContentService.createTextOutput(JSON.stringify({status:"error",message:"Missing params"}))
    .setMimeType(ContentService.MimeType.JSON)
    .setHeaders({"Access-Control-Allow-Origin":"*"});
}
